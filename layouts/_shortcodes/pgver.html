{{- /* The PostgreSQL major-version support strip: five numbers, coloured by whether the extension is available for that major. It is the fastest read on a catalog row, so it stays a run of plain spans with class names instead of inline colour — the same strip is drawn thousands of times per page. @param {string} 0 comma-separated majors, e.g. "18,17,16,15,14" @param {string} 1 comma-separated codes: r red, g green, y yellow, b blue, o orange, gr grey @example pgver "18,17,16,15,14" "g,g,g,r,r" */ -}} {{- $versions := split (trim (.Get 0) " ") "," -}} {{- $codes := split (trim (.Get 1) " ") "," -}} {{- $names := dict "r" "red" "g" "green" "y" "yellow" "b" "blue" "o" "orange" "gr" "gray" -}} {{- range $i, $v := $versions -}} {{- $v = trim $v " " -}} {{- $code := "gr" -}} {{- if lt $i (len $codes) }}{{ $code = trim (index $codes $i) " " }}{{ end -}} {{- $name := index $names $code | default "gray" -}} {{ $v }} {{- end -}}
